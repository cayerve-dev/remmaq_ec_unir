--Eliminar contenedor docker
docker compose down
--Levantar contenedor docker
docker compose up -d --build
--Verificar contenedor docker
docker ps


--Instalar 7z
docker exec -u 0 nifi bash -lc "apt-get update && apt-get install -y --no-install-recommends 7zip unrar && if [ ! -x /usr/bin/7z ] && [ -x /usr/bin/7zz ]; then ln -s /usr/bin/7zz /usr/bin/7z; fi && rm -rf /var/lib/apt/lists/*"
